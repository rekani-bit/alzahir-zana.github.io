CREATE TABLE products ( id BIGINT PRIMARY KEY AUTO_INCREMENT, category_id BIGINT, brand_id BIGINT, unit_id BIGINT, barcode VARCHAR(100) UNIQUE, name VARCHAR(200) NOT NULL, cost_price DECIMAL(12,2) NOT NULL, selling_price DECIMAL(12,2) NOT NULL, min_stock INT DEFAULT 0, expires_at DATE NULL, FOREIGN KEY(category_id) REFERENCES categories(id), FOREIGN KEY(brand_id) REFERENCES brands(id), FOREIGN KEY(unit_id) REFERENCES units(id) ); CREATE TABLE inventories ( id BIGINT PRIMARY KEY AUTO_INCREMENT, branch_id BIGINT, product_id BIGINT, quantity DECIMAL(12,3) DEFAULT 0, UNIQUE(branch_id, product_id), FOREIGN KEY(branch_id) REFERENCES branches(id), FOREIGN KEY(product_id) REFERENCES products(id) ); -- Seed data INSERT INTO roles(name) VALUES ('Super Admin'), ('Manager'), ('Cashier'); INSERT INTO branches(name) VALUES ('Main Branch');
